function maskCPF(v) {
  v = v.replace(/\D/g,"").slice(0,11);
  if(v.length>9) return v.replace(/(\d{3})(\d{3})(\d{3})(\d{1,2})/,"$1.$2.$3-$4");
  if(v.length>6) return v.replace(/(\d{3})(\d{3})(\d{1,3})/,"$1.$2.$3");
  if(v.length>3) return v.replace(/(\d{3})(\d{1,3})/,"$1.$2");
  return v;
}
function maskPhone(v) {
  v = v.replace(/\D/g,"").slice(0,11);
  if(v.length>10) return v.replace(/(\d{2})(\d{5})(\d{4})/,"($1) $2-$3");
  if(v.length>6)  return v.replace(/(\d{2})(\d{4,5})(\d{0,4})/,"($1) $2-$3");
  if(v.length>2)  return v.replace(/(\d{2})(\d{0,5})/,"($1) $2");
  return v;
}

  if(cat.includes("IMEDIATO"))       return { fg:"#ef4444", bg:"rgba(239,68,68,0.12)",  bd:"rgba(239,68,68,0.3)"  };
  if(cat.includes("MUITO URGENTE"))  return { fg:"#f97316", bg:"rgba(249,115,22,0.12)", bd:"rgba(249,115,22,0.3)" };
  if(cat.includes("URGENTE"))        return { fg:"#eab308", bg:"rgba(234,179,8,0.12)",  bd:"rgba(234,179,8,0.3)"  };
  if(cat.includes("POUCO URGENTE"))  return { fg:"#22c55e", bg:"rgba(34,197,94,0.12)",  bd:"rgba(34,197,94,0.3)"  };
  if(cat.includes("NÃO URGENTE"))    return { fg:"#3b82f6", bg:"rgba(59,130,246,0.12)", bd:"rgba(59,130,246,0.3)" };
  return                               { fg:"#94a3b8", bg:"rgba(148,163,184,0.12)",     bd:"rgba(148,163,184,0.3)"};
}

function FInput({ value, onChange, placeholder, type="text" }) {
  const [f,setF] = useState(false);

function PortalPaciente({ onPacienteCadastrado }) {
  const [step, setStep] = useState(PATIENT_STEPS.CONSENT);
  const [formData, setFormData] = useState(null);

  function handleFormSubmit(data) {
    setFormData(data);
    setStep(PATIENT_STEPS.TRIAGE);
  }

  function handleConcluir(prontuario) {
    onPacienteCadastrado(prontuario);
    setStep(PATIENT_STEPS.CONSENT);
    setFormData(null);
    alert(`✅ ${prontuario.nome} adicionado à fila médica com classificação: ${prontuario.manchester}`);
  }

  return(
    <div style={{maxWidth:700,margin:"0 auto",padding:"28px 16px 60px"}}>
      {/* Indicador de progresso */}
      <div style={{display:"flex",alignItems:"center",gap:8,marginBottom:24}}>
        {[["1","Consentimento",PATIENT_STEPS.CONSENT],["2","Cadastro",PATIENT_STEPS.FORM],["3","Triagem IA",PATIENT_STEPS.TRIAGE]].map(([n,label,s],i)=>(
          <div key={s} style={{display:"flex",alignItems:"center",gap:8,flex:1}}>
            <div style={{display:"flex",alignItems:"center",gap:6}}>
              <div style={{
                width:26,height:26,borderRadius:"50%",display:"flex",alignItems:"center",
                justifyContent:"center",fontSize:12,fontWeight:700,
                background:step===s?"#2563eb":step===PATIENT_STEPS.TRIAGE&&s!==PATIENT_STEPS.TRIAGE?"rgba(37,99,235,0.3)":step===PATIENT_STEPS.FORM&&s===PATIENT_STEPS.CONSENT?"rgba(37,99,235,0.3)":"rgba(255,255,255,0.08)",
                color:step===s?"white":"rgba(255,255,255,0.4)",
                border:step===s?"none":"1px solid rgba(255,255,255,0.1)",
              }}>{n}</div>
              <span style={{fontSize:12,color:step===s?"white":"rgba(255,255,255,0.3)",fontWeight:step===s?600:400}}>{label}</span>
            </div>
            {i<2&&<div style={{flex:1,height:1,background:"rgba(255,255,255,0.08)"}}/>}
          </div>
        ))}
      </div>

      {step===PATIENT_STEPS.CONSENT && <ConsentStep onNext={()=>setStep(PATIENT_STEPS.FORM)}/>}
      {step===PATIENT_STEPS.FORM    && <FormStep onSubmit={handleFormSubmit}/>}
      {step===PATIENT_STEPS.TRIAGE  && <TriageStep data={formData} onConcluir={handleConcluir}/>}
    </div>
  );
}

                boxShadow: tab===t.id?"0 2px 12px rgba(37,99,235,0.35)":"none",
              }}>
                <span style={{marginRight:6}}>{t.icon}</span>{t.label}
                {t.badge>0&&(
                  <span style={{
                    position:"absolute",top:-4,right:-4,width:18,height:18,borderRadius:"50%",
                    background:"#ef4444",color:"white",fontSize:10,fontWeight:700,

          {/* Status */}
          <div style={{display:"flex",alignItems:"center",gap:6,padding:"6px 14px",

              display:"inline-block",animation:"pulse 2s infinite"}}/>

          </div>
        </div>
      </nav>

      {/* Conteúdo */}
      {tab===TABS.PACIENTE && <PortalPaciente onPacienteCadastrado={handlePacienteCadastrado}/>}
      {tab===TABS.MEDICO   && <PainelMedico  fila={fila} onConsultaFinalizada={handleConsultaFinalizada}/>}
      {tab===TABS.FINTECH  && <ModuloFintech transacoes={transacoes} lucroTotal={lucro}/>}

      <style>{`

      `}</style>
    </div>
  );
